SELECT line_items.order_id, SUM(price * quantity) AS total_price FROM products JOIN line_items ON products.product_id = line_items.product_id JOIN orders ON line_items.order_id = orders.order_id GROUP BY line_items.order_id ORDER BY line_items.order_id ASC LIMIT 5;

SELECT customer_name, AVG(t.total_price) AS average_order_price FROM customers c JOIN (
	SELECT customer_id, SUM(price * quantity) AS total_price FROM products p JOIN line_items l ON p.product_id = l.product_id JOIN orders AS o ON l.order_id = o.order_id GROUP BY l.order_id, o.customer_id
) AS t ON c.customer_id = t.customer_id GROUP BY c.customer_id, c.customer_name ORDER BY c.customer_name ASC;
